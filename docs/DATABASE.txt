  // đúng MỘT chứng từ nguồn có giá trị, khớp với payment_type
  sale_invoice_id uuid
  rental_slip_id uuid
  stock_receipt_id uuid [note: 'Chi tiền thu mua sách cũ']
  return_id uuid [note: 'Hoàn cọc']
  work_record_id uuid [note: 'Chi trả lương cho một bản ghi ca làm']
  cash_received bigint [note: 'Tiền khách đưa (tiền mặt); tiền thối = cash_received - amount']
  qr_reference varchar(60) [unique, note: 'Nội dung chuyển khoản duy nhất (VietQR)']
  qr_expires_at timestamptz
  bank_reference varchar(80) [note: 'Mã giao dịch ngân hàng khi chi chuyển khoản']
  created_by uuid [not null]
  created_at timestamptz [not null, default: `now()`]
  paid_at timestamptz
  Note: 'Mọi giao dịch tiền (vào/ra) trong một bảng, mỗi giao dịch gắn đúng một chứng từ. return_charge ghi nhận khoản thu thêm khi cọc không đủ; giao dịch thành công bị đóng băng và sửa sai bằng giao dịch đảo.'
}
